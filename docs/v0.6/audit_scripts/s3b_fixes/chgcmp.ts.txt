import { readFileSync } from "node:fs";
const [a, b] = process.argv.slice(2).map((l) => JSON.parse(readFileSync(`${import.meta.dirname}/chg_${l}.json`, "utf8")));
const t: Record<string, Record<string, [number, number]>> = {};
for (const id of Object.keys(a)) {
  const [ck, xk, ...ha] = a[id].split(","); const hb = b[id].split(",").slice(2);
  ["ctx", "ctxGroup", "sentence", "xp"].forEach((k, i) => {
    const kind = k === "xp" ? xk : ck;
    const r = ((t[k] ??= {})[kind] ??= [0, 0]); r[0]++; if (ha[i] !== hb[i]) r[1]++;
  });
}
for (const [k, v] of Object.entries(t)) console.log(k, Object.entries(v).map(([m, [n, d]]) => `${m}: ${d}/${n} differ`).join("; "));
