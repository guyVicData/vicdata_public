// Which focus items do the differing scenarios have? (node focuscheck.ts a b)
import { readFileSync } from "node:fs";
const [a, b] = process.argv.slice(2).map((l) => JSON.parse(readFileSync(`${import.meta.dirname}/out_${l}.json`, "utf8")));
const foci = new Map<string, number>();
for (const id of Object.keys(a)) if (id.split("/").length > 3 && id in b && a[id] !== b[id]) { const f = id.split("/").slice(0, 3).join("/"); foci.set(f, (foci.get(f) ?? 0) + 1); }
for (const [f, n] of foci) console.log(n, f);
