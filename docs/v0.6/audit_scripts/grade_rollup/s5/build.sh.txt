#!/bin/zsh
# build.sh <br|main>: CSS + bundle from that tree.
S=/private/tmp/claude-501/-Users-guy-dev-vicdata-public/22f4c365-7069-4ac9-ab6f-8e923021fb87/scratchpad/s5g
R=$([ "$1" = main ] && echo $S/main || echo /Users/guy/dev/vicdata_public)

echo built $1
