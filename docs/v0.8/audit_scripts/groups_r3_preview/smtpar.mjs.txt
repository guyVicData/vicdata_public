// SMT (and the Teacher home) parity: the same fixtures drawn by the base and the branch bundles, every
// dashboard x school x theme x width (+ the home), each panel's default view and then every rail view of
// the large panel; full-page PNG and #teacher-root's DOM, compared byte for byte.
//   node smtpar.mjs <port0> <workers>   (harness on 127.0.0.1:$HTTP_PORT)
import { spawn } from "node:child_process";
import { mkdirSync, writeFileSync } from "node:fs";
const [port0, workers] = process.argv.slice(2);
const OUT = process.env.OUT ?? "smtpar_out";
const cases = [];
for (const urn of ["137625", "110110"]) for (const page of ["overview", "academic", "pupils", "context", "home"]) for (const theme of ["dark", "light"]) for (const w of [1440, 390]) cases.push({ urn, page, theme, w });
const TREES = (process.env.TREES ?? "base,br").split(",");
const jobs = cases.flatMap((c) => TREES.map((t) => ({ ...c, tree: t })));
const sleep = (ms) => new Promise((r) => setTimeout(r, ms));
async function browser(port) {
  const chrome = spawn("/Applications/Google Chrome.app/Contents/MacOS/Google Chrome", ["--headless=new", `--remote-debugging-port=${port}`, "--no-first-run", "--disable-gpu", "--hide-scrollbars", "--force-color-profile=srgb", "--font-render-hinting=none", `--user-data-dir=/tmp/smtpar-${port}-${Date.now()}`, "about:blank"], { stdio: "ignore" });
  let target;
  for (let i = 0; i < 60 && !target; i++) { await sleep(200); try { target = (await (await fetch(`http://127.0.0.1:${port}/json/list`)).json()).find((t) => t.type === "page"); } catch {} }
  const ws = new WebSocket(target.webSocketDebuggerUrl);
  await new Promise((r) => (ws.onopen = r));
  let id = 0; const pend = new Map();
  ws.onmessage = (m) => { const d = JSON.parse(m.data); if (d.id && pend.has(d.id)) { pend.get(d.id)(d); pend.delete(d.id); } };
  const send = (method, params = {}) => new Promise((r) => { const i = ++id; pend.set(i, r); ws.send(JSON.stringify({ id: i, method, params })); });
  await send("Network.enable"); await send("Network.setBlockedURLs", { urls: ["*basemaps*", "*cartocdn*", "*tile.openstreetmap*", "*tiles*"] }); await send("Page.enable"); await send("Runtime.enable");
  return { send, close: () => { ws.close(); chrome.kill(); } };
}
async function capture(b, j) {
  const dir = `${OUT}/${j.tree}`; mkdirSync(dir, { recursive: true });
  const name = `${j.urn}_${j.page}_${j.theme}_${j.w}`;
  await b.send("Emulation.setDeviceMetricsOverride", { width: j.w, height: 1300, deviceScaleFactor: 1, mobile: j.w < 500 });
  const url = `http://127.0.0.1:${process.env.HTTP_PORT}/index.html?case=${j.urn}&page=${j.page}&theme=${j.theme}&bundle=bundle_${j.tree}&css=app_${j.tree}`;
  await b.send("Page.navigate", { url }); await sleep(4500);
  const shoot = async (step) => {
    const h = await b.send("Runtime.evaluate", { expression: "Math.min(5000, Math.max(document.documentElement.scrollHeight, document.body.scrollHeight))", returnByValue: true });
    await b.send("Emulation.setDeviceMetricsOverride", { width: j.w, height: h.result.result.value, deviceScaleFactor: 1, mobile: j.w < 500 }); await sleep(500);
    const dom = await b.send("Runtime.evaluate", { expression: "(document.getElementById('teacher-root')||document.body).outerHTML", returnByValue: true });
    const png = await b.send("Page.captureScreenshot", { format: "png" });
    writeFileSync(`${dir}/${name}_s${step}.html`, dom.result.result.value ?? "");
    writeFileSync(`${dir}/${name}_s${step}.png`, Buffer.from(png.result.data, "base64"));
  };
  await shoot(0);
  // The large panel's rail (desktop): each view in turn.
  if (j.w > 500 && j.page !== "home") {
    const RAIL = "[...document.querySelectorAll('button[aria-pressed]')].filter((x) => x.className.includes('h-[26px]'))";
    const n = await b.send("Runtime.evaluate", { expression: `${RAIL}.length`, returnByValue: true });
    for (let i = 0; i < Math.min(14, n.result.result.value ?? 0); i++) {
      await b.send("Runtime.evaluate", { expression: `(${RAIL}[${i}]||{click(){}}).click()` }); await sleep(1300);
      await shoot(i + 1);
    }
  }
}
let next = 0; const t0 = Date.now();
await Promise.all(Array.from({ length: Number(workers) }, async (_, k) => {
  const b = await browser(Number(port0) + k);
  while (next < jobs.length) { const j = jobs[next++]; try { await capture(b, j); } catch (e) { console.log("ERR", j, e.message); } if (next % 10 === 0) console.log(`[smtpar +${Math.round((Date.now() - t0) / 1000)}s] ${next}/${jobs.length}`); }
  b.close();
}));
console.log("done", jobs.length);
process.exit(0);
