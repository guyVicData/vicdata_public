// SMT fixtures for a Harris school: as its own SMT sees it (caller null, as the SMT preview), and as head
// office sees it (no share: the Admissions panels "Private to {school}", as smtCaller gives head office).
import { writeFileSync } from "node:fs";
import { smtSchool } from "/Users/guy/dev/vicdata_public/src/lib/smt/school";
import { buildOverview } from "/Users/guy/dev/vicdata_public/src/lib/smt/overview";
import { buildPupils } from "/Users/guy/dev/vicdata_public/src/lib/smt/pupils";
import { resolveDefaultNearest } from "/Users/guy/dev/vicdata_public/src/lib/chooser-sets";
import { setPhaseOf, levelChips } from "/Users/guy/dev/vicdata_public/src/lib/smt/levels";
import { groupNote } from "/Users/guy/dev/vicdata_public/src/catalogue/notes";
const urn = process.argv[3];
const s = (await smtSchool(urn))!;
const near = await resolveDefaultNearest(urn, setPhaseOf(s.phases));
const set = { kind: "nearest", label: "10 nearest schools", urns: near.rows.map((r: { urn: string }) => r.urn).filter((u: string) => u !== urn) };
const out: Record<string, unknown> = {};
for (const [key, caller] of [[urn, null], [`${urn}ho`, { supabase: null, privateNote: groupNote("admissionsPrivate", { school: s.name }) }]] as const) {
  const pupils: Record<string, unknown> = {};
  for (const l of levelChips(s.phases)) pupils[`${l}|all|all`] = await buildPupils(s, set, { sex: "all", boarding: "all" }, { phase: l, area: null }, caller);
  out[key] = { urn, name: s.name, school: s, nearest: { rows: near.rows.map((r: { urn: string; name?: string }) => ({ urn: r.urn, name: r.name })) }, overview: { "all|all": await buildOverview(s, set, { sex: "all", boarding: "all" }, caller) }, academic: {}, pupils, context: {} };
}
writeFileSync(process.argv[2], `window.__SMTFX = ${JSON.stringify(out)};\n`);
console.log("wrote", Object.keys(out));
