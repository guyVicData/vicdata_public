window.fetch = async (input, init) => {
  const u = new URL(typeof input === "string" ? input : input.url, location.href);
  const json = (b, s = 200) => new Response(JSON.stringify(b), { status: s, headers: { "Content-Type": "application/json" } });
  if (!u.pathname.startsWith("/api/")) return realFetch(input, init);
  const q = u.searchParams;
  if (u.pathname === "/api/usage") return new Response(null, { status: 204 });
  if (u.pathname === "/api/group/dash") {
    const k = `${q.get("dash")}|${q.get("reading")}|${q.get("focus")}|${q.get("compare") || ""}`;
    const r = FX.dash[k];
    if (!r) return json({ error: "no fixture " + k }, 500);
    return json({ ...r, viewer: { ...r.viewer, as: WHO, urn: WHO === "school" ? (r.map[0] || {}).urn : null }, sets: forWho(r.sets) });
  }
  if (u.pathname === "/api/group/sets") {
    if (WHO === "school") return json({ error: "Not found." }, 404);
    const base = { ...FX.sets, viewer: { ...FX.sets.viewer, as: WHO }, can: WHO === "head_office" ? FX.sets.can : { keep: false, admin: false }, sets: WHO === "head_office" ? FX.sets.sets : FX.sets.sets.filter((s) => s.scope === "group") };
    return json({ ...base, ...(q.get("with") === "cards" ? { cards: FX.cards } : {}), ...(q.get("with") === "schools" ? { schools: FX.schools } : {}) });
  }
  if (u.pathname === "/api/group/library/group") return FX.libpages && FX.libpages[q.get("id")] ? json(FX.libpages[q.get("id")]) : json({ error: "Not found." }, 404);
  if (u.pathname === "/api/group/library") return json(FX.library);
  if (u.pathname === "/api/group/projection") return json({ rows: [] });
  if (u.pathname === "/api/group/me") return json({ offices: [], viewAs: false });
  if (u.pathname === "/api/teacher/phases") return json({ phases: [] });
  return json({ error: "unstubbed " + u.pathname }, 404);
};
