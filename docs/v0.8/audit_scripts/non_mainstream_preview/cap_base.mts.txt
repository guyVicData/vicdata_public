// Group sets harness fixtures: the dash route's own steps (resolveCompare + the builders) over the
// fixtures, with a few stored sets as the head office would have them. Run with the fixture switch on.
import { writeFileSync } from "node:fs";
const R = "$SCRATCH/base";
const A = await import(`${R}/src/lib/groups/aggregates.ts`);
const D = await import(`${R}/src/lib/groups/dash.ts`);
const R3 = await import(`${R}/src/lib/groups/dash-r3.ts`);
const GS = await import(`${R}/src/lib/groups/group-sets.ts`);
const { groupMembers } = await import(`${R}/src/lib/groups/production.ts`);
if (A.groupDataSource() !== "fixtures") throw new Error("the fixture switch is off");
const USER = "00000000-0000-0000-0000-000000000001";
const set = (id: string, name: string, kind: string, scope: string, members: string[], o: Record<string, unknown> = {}) => ({ id, name, description: null, kind, scope, isDefault: false, pinned: false, archived: false, ownerProfileId: scope === "personal" ? USER : null, members, updatedAt: "2026-11-12T10:00:00Z", ...o });
const index = await A.groupLibraryIndex();
const G = [
  { key: "harris", groupId: "3320", account: "a0000000-0000-4000-8000-000000003320", name: "Harris Federation", sets: [
    set("a1111111-1111-4111-8111-111111111111", "London and the South: big MATs", "group_peers", "group", ["2157", "5143", "3586", "4124", "4076"]),
    set("b2222222-2222-4222-8222-222222222222", "Ark, United Learning and a foundation trust", "group_peers", "group", ["2157", "5143", "1337"]),
    set("c3333333-3333-4333-8333-333333333333", "Nearest five (mine)", "group_peers", "personal", ["4076", "2053", "4474", "4127", "4320"]),
    set("e5555555-5555-4555-8555-555555555555", "Harris secondaries, first 18", "group_schools", "group", ["131747","131897","132711","134225","134226","135249","135311","135677","135816","135955","136579","136861","137121","137379","137549","138449","138495","139181"]),
    set("f6666666-6666-4666-8666-666666666666", "My two Croydon primaries", "group_schools", "personal", ["131747","134225"]),
  ] },
  { key: "dukes", groupId: "club_iss:dukes-education", account: "a0000000-0000-4000-8000-0000000d0e55", name: "Dukes Education", sets: [
    set("d4444444-4444-4444-8444-444444444444", "Independent groups we watch", "group_peers", "group", ["ch:01550681", "ch:16134999", "club_iss:cognita", "club_iss:inspired-education", "club_iss:girls-day-school-trust"]),
  ] },
];
const out: Record<string, unknown> = {};
for (const g of G) {
  const members = await groupMembers(g.groupId).catch(() => []);
  const membership = members.map((m: any) => ({ urn: m.urn, name: m.name, confidence: m.confidence, joined: m.joined, lastSeen: m.lastSeen, source: m.source }));
  const schools = await A.groupSchoolsYear(g.groupId);
  const mapOf = (census: number) => schools.filter((s: any) => s.role === "member" && s.years.some((y: any) => y.listed_now)).map((s: any) => { const f = s.years.find((y: any) => y.period === census)?.figures ?? null; return { urn: s.urn, name: s.name ?? s.urn, easting: s.easting, northing: s.northing, roll: f?.roll ?? null, trend: f?.trend["1"] ?? null, outside: s.outside_england, likely: s.confidence === "likely", phases: f?.phases ?? [] }; });
  const viewer = { as: "head_office", accountId: g.account, groupId: g.groupId, name: g.name, urn: null };
  const sets = { sets: g.sets, notSetUp: false, personalActive: g.sets.filter((s) => s.scope === "personal").length + 1, cap: 8, admin: true, canKeep: true, defaults: { group_peers: null, group_schools: null } };
  const card = new Map(index.groups.map((c: any) => [c.group_id, c]));
  const meta = (id: string) => { const c: any = card.get(id); return { id, label: c ? D.displayName(c.name) : id, type: c?.type ?? "unknown", schools: c?.schools ?? 0 }; };
  const resp: Record<string, unknown> = {};
  for (const reading of ["at_time", "listed_now"]) {
    const agg = await A.groupAggregate(g.groupId, reading);
    const sector = agg.group.type.startsWith("independent") ? "independent" : "state";
    const us = index.groups.find((c: any) => c.group_id === g.groupId);
    const nearest = GS.nearestInSize(index.groups, us).map((c: any) => meta(c.group_id));
    const keys = ["", agg.peer?.set_id, `non_group:${sector}`, "sat:all", "areas:la", "england:all", "nearest", ...g.sets.map((s) => `set:${s.id}`), g.key === "harris" ? "peer_named:2157,5143" : "peer_named:ch:01550681"];
    for (const focus of ["all", "primary", "secondary", "post16"]) {
      if (focus !== "all" && !agg.rows.some((r: any) => r.phase === focus && r.figures.n_schools > 0)) continue;
      for (const raw of keys) {
        if (reading === "listed_now" && raw !== "" ) continue;
        const { key, compare, groupSet } = GS.resolveCompare(raw || null, { bandKey: agg.peer?.set_id ?? null, sector, ourId: g.groupId, sets: g.sets, defaultId: null, nearest, meta });
        const comps = await A.groupComparisons(g.groupId, D.kindsFor(compare));
        const pass = { groupSet, addressKey: key, nearest };
        const base = { viewer, source: "fixtures", membership, sets, nearest };
        const o = D.buildOverview({ agg, comps, compare, focus, ...pass });
        resp[`overview|${reading}|${focus}|${raw}`] = { ...base, overview: o, map: mapOf(o.years.census) };
        const input = { agg, comps, schools, compare, focus, ...pass };
        const a = R3.buildAcademic(input);
        resp[`academic|${reading}|${focus}|${raw}`] = { ...base, academic: a, now: a.overview, overview: a.overview, map: mapOf(a.overview.years.census) };
        const p = R3.buildPupils(input);
        resp[`pupils|${reading}|${focus}|${raw}`] = { ...base, pupils: p, now: p.overview, overview: p.overview, map: mapOf(p.overview.years.census) };
      }
      const now = D.buildOverview({ agg, comps: [], compare: "peer_band", focus });
      resp[`schools|${reading}|${focus}|`] = { viewer, source: "fixtures", membership, sets, nearest, now, map: mapOf(now.years.census), schools: D.buildSchools({ agg, schools, focus, members, vicdata: new Set(), headOffice: true }) };
    }
  }
  // /api/group/sets (the editor, Manage sets, Add to set…)
  const rows = schools.filter((s: any) => s.role === "member").map((s: any) => { const latest = [...s.years].reverse().find((y: any) => y.figures)?.figures ?? null; return { urn: s.urn, name: s.name ?? s.urn, la: s.la_name, phases: latest?.phases ?? [], roll: latest?.roll ?? null, likely: s.confidence === "likely", current: s.years.some((y: any) => y.listed_now), counted: latest?.counted !== false }; }).sort((a: any, b: any) => Number(b.current) - Number(a.current) || a.name.localeCompare(b.name));
  const us = index.groups.find((c: any) => c.group_id === g.groupId) ?? null;
  const setsResp = { viewer: { as: "head_office", accountId: g.account, groupId: g.groupId, name: g.name, userId: USER }, source: "fixtures", can: { keep: true, admin: true }, sets: g.sets, notSetUp: false, personalActive: sets.personalActive, cap: 8, defaults: sets.defaults, us };
  out[g.key] = { dash: resp, sets: setsResp, cards: index.groups.filter((c: any) => c.group_id !== g.groupId), schools: rows, library: { viewer: { as: "head_office", accountId: g.account, groupId: g.groupId, name: g.name, admin: true }, source: "fixtures", build: index.build, us, cards: index.groups.filter((c: any) => c.group_id !== g.groupId) } };
  console.log(g.key, Object.keys(resp).length, "dash responses");
}
writeFileSync(process.argv[2] ?? "fx/gs.js", `window.__GS = ${JSON.stringify(out)};\n`);
