#!/bin/zsh
# The round's screenshots: ./shots.sh <outdir> [port]   (the branch harness served on 127.0.0.1:<port>)
H=${0:A:h}; OUT=$1; PORT=${2:-3191}
HA="g=harris&account=a0000000-0000-4000-8000-000000003320"; NX="g=nexus&account=a0000000-0000-4000-8000-000000016125"
BH="g=birmingham&account=a0000000-0000-4000-8000-000000017126"; SH="g=shaw&account=a0000000-0000-4000-8000-000000004499"
RT="returnTo=%2Fteacher%2Fgroup%2Foverview"
btn() { echo "[...document.querySelectorAll('button')].find((b)=>b.getAttribute('aria-label')==='$1'||b.textContent.trim()==='$1').click()"; }
shot() { # name query width theme [click] [nofull] [height]
  CLICK="$5" NOFULL="$6" node $H/pshot.mjs $OUT/$1-$3-$4.png "http://127.0.0.1:$PORT/index.html?$2&theme=$4" $3 ${7:-1000} 4500 | sed "s#^#  #"
}
for t in dark light mid; do for w in 1440 390; do
  h=$([ $w = 390 ] && echo 844 || echo 1000)
  shot 01-library-special-ap-nursery-section-harris "page=library&$HA&section=nm" $w $t "document.querySelector('[data-nm-intro]').open=true"
  shot 02-library-group-page-nexus "page=libgroup&id=16125&$HA" $w $t
  shot 03-overview-nexus "page=overview&$NX" $w $t
  shot 04-mainstream-line-nexus "page=overview&$NX" $w $t "$(btn 'Mainstream line')"
  shot 05-academic-no-results-nexus "page=academic&$NX" $w $t
  shot 06-schools-nexus "page=schools&$NX" $w $t
  shot 07-side-by-side-refused-nexus "page=overview&$NX&compare=set:b8888888-8888-4888-8888-888888888888" $w $t
  shot 08-overview-nursery-birmingham "page=overview&$BH" $w $t
  shot 09-schools-nursery-birmingham "page=schools&$BH" $w $t
  shot 10-add-to-set-refused-nexus-library "page=library&$NX&section=mainstream" $w $t "document.querySelector('[data-add-to-set] button').click()" 1 $h
  shot 11-editor-refuses-ark-nexus "page=edit&$NX&kind=group_peers&set=b8888888-8888-4888-8888-888888888888&$RT" $w $t "document.querySelector('[data-add-groups]').click()" 1 $h
  shot 12-schools-placing-shaw "page=schools&$SH" $w $t "$(btn 'Full screen: Schools')" 1 $([ $w = 390 ] && echo 1600 || echo 1700)
  shot 13-schools-pooled-line-shaw "page=schools&$SH" $w $t
done; done
