// 0.8 feedback C: the walks' and homes' screenshot harness (builds A and B), plus the real dashboards:
// ?page=smtdash&dash=overview|academic|pupils|context, ?page=admdash. ?walk=…&step=N (1-based)&theme=,
// ?page=home&s=ks4,smt,admissions. Fed the routes' real answers (fixtures captured read-only).
import { createRoot } from "react-dom/client";
import { WalkPage } from "@/components/front-door/WalkPage";
import { HomeFrame } from "@/components/front-door/home";
import { AdmissionsSection, SmtSection, TeacherPhaseSection } from "@/components/front-door/sections";
import { SmtDashboard } from "@/components/smt/SmtDashboard";
import { AdmissionsDashboard } from "@/components/admissions/AdmissionsDashboard";
const P = new URLSearchParams(location.search);
const URN = P.get("urn") ?? "110110";
const SECTIONS: Record<string, () => JSX.Element> = {
  ks4: () => <TeacherPhaseSection key="ks4" urn={URN} phase="ks4" />,
  ks5: () => <TeacherPhaseSection key="ks5" urn={URN} phase="ks5" />,
  smt: () => <SmtSection key="smt" />,
  admissions: () => <AdmissionsSection key="adm" urn={URN} />,
};
const pg = P.get("page");
const page =
  pg === "home" ? <HomeFrame heading={null} sections={(P.get("s") ?? "smt").split(",").map((k) => SECTIONS[k]())} also={[]} />
  : pg === "smtdash" ? <SmtDashboard dash={(P.get("dash") ?? "overview") as "overview"} />
  : pg === "admdash" ? <AdmissionsDashboard />
  : <WalkPage walkId={P.get("walk") ?? "gcse-many"} />;
createRoot(document.getElementById("root")!).render(page);
