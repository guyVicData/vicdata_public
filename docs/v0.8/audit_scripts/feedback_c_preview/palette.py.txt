# Review fixes (14 Nov, M7): pick the eight subject-family colours. CIEDE2000 between every pair (min, logged),
# away from the product accents and fixed lines, each >= 3:1 against its plate (WCAG 1.4.11 graphics).
import math, colorsys, itertools, json, sys
def srgb2lin(c): c/=255; return c/12.92 if c<=0.04045 else ((c+0.055)/1.055)**2.4
def hex2rgb(h): h=h.lstrip('#'); return tuple(int(h[i:i+2],16) for i in (0,2,4))
def rgb2hex(r): return '#'+''.join(f'{max(0,min(255,round(x))):02x}' for x in r)
def lum(h): r,g,b=[srgb2lin(c) for c in hex2rgb(h)]; return 0.2126*r+0.7152*g+0.0722*b
def contrast(a,b): la,lb=sorted([lum(a),lum(b)],reverse=True); return (la+0.05)/(lb+0.05)
def lab(h):
    r,g,b=[srgb2lin(c) for c in hex2rgb(h)]
    x=(0.4124*r+0.3576*g+0.1805*b)/0.95047; y=(0.2126*r+0.7152*g+0.0722*b); z=(0.0193*r+0.1192*g+0.9505*b)/1.08883
    f=lambda t: t**(1/3) if t>0.008856 else 7.787*t+16/116
    fx,fy,fz=f(x),f(y),f(z); return (116*fy-16, 500*(fx-fy), 200*(fy-fz))
def de2000(h1,h2):
    L1,a1,b1=lab(h1); L2,a2,b2=lab(h2)
    C1=math.hypot(a1,b1); C2=math.hypot(a2,b2); Cb=(C1+C2)/2
    G=0.5*(1-math.sqrt(Cb**7/(Cb**7+25**7)))
    a1p=(1+G)*a1; a2p=(1+G)*a2; C1p=math.hypot(a1p,b1); C2p=math.hypot(a2p,b2)
    h1p=math.degrees(math.atan2(b1,a1p))%360; h2p=math.degrees(math.atan2(b2,a2p))%360
    dL=L2-L1; dC=C2p-C1p
    dh=h2p-h1p
    if C1p*C2p==0: dh=0
    elif dh>180: dh-=360
    elif dh<-180: dh+=360
    dH=2*math.sqrt(C1p*C2p)*math.sin(math.radians(dh/2))
    Lb=(L1+L2)/2; Cbp=(C1p+C2p)/2
    hb=(h1p+h2p)/2 if abs(h1p-h2p)<=180 else (h1p+h2p+360)/2
    if C1p*C2p==0: hb=h1p+h2p
    T=1-0.17*math.cos(math.radians(hb-30))+0.24*math.cos(math.radians(2*hb))+0.32*math.cos(math.radians(3*hb+6))-0.2*math.cos(math.radians(4*hb-63))
    dth=30*math.exp(-((hb-275)/25)**2); Rc=2*math.sqrt(Cbp**7/(Cbp**7+25**7))
    Sl=1+0.015*(Lb-50)**2/math.sqrt(20+(Lb-50)**2); Sc=1+0.045*Cbp; Sh=1+0.015*Cbp*T; Rt=-math.sin(math.radians(2*dth))*Rc
    return math.sqrt((dL/Sl)**2+(dC/Sc)**2+(dH/Sh)**2+Rt*(dC/Sc)*(dH/Sh))
RES_LIGHT={"GCSE green":"#12a05c","Post-16 violet":"#7c5ce6","SMT orange":"#c2410c","Admissions sky":"#0b6fa6","England":"#a86f00","LA":"#3987e5","Nearest schools":"#d55181","Focus school red":"#dc2626","Group bronze":"#8a5a2b"}
RES_DARK={"GCSE green":"#34d399","Post-16 violet":"#a78bfa","SMT orange":"#fb923c","Admissions sky":"#22d3ee","England":"#c98500","LA":"#3987e5","Nearest schools":"#d55181","Focus school red":"#dc2626","Group bronze":"#a8743f"}
PLATE={"light":"#ffffff","dark":"#0b0b0d"}
def cands(theme):
    out=[]
    for hue in range(0,360,5):
        for s in (0.35,0.5,0.65,0.8):
            for l in ((0.28,0.33,0.38,0.43,0.48) if theme=="light" else (0.6,0.66,0.72,0.78)):
                r,g,b=colorsys.hls_to_rgb(hue/360,l,s); h=rgb2hex((r*255,g*255,b*255))
                if contrast(h,PLATE[theme])>=3.0: out.append(h)
    return out
def pick(theme, res, minres, seed=None):
    cs=[c for c in cands(theme) if min(de2000(c,v) for v in res.values())>=minres]
    best=None
    # greedy max-min from each starting colour, keep the best set
    for start in (seed or cs[::7]):
        chosen=[start]
        while len(chosen)<8:
            nxt=max(cs,key=lambda c: min(de2000(c,x) for x in chosen))
            chosen.append(nxt)
        m=min(de2000(a,b) for a,b in itertools.combinations(chosen,2))
        if not best or m>best[0]: best=(m,chosen)
    return best,len(cs)
if __name__=="__main__":
    out={}
    for theme,res in (("light",RES_LIGHT),("dark",RES_DARK)):
        for minres in (14,12,10,8):
            best,n=pick(theme,res,minres)
            if best[0]>=14: break
        out[theme]={"minPair":round(best[0],1),"minToReserved":minres,"colours":best[1],"candidates":n}
        print(theme, out[theme])
    json.dump(out,open(sys.argv[1],"w"),indent=1)
