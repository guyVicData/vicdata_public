# pxdiff.py <dirA> <dirB> <png>...: size, differing pixel count, max channel step, bbox
import sys
from PIL import Image, ImageChops
a,b=sys.argv[1:3]
for f in sys.argv[3:]:
    x=Image.open(f"{a}/{f}").convert("RGB"); y=Image.open(f"{b}/{f}").convert("RGB")
    if x.size!=y.size: print(f,"size",x.size,y.size); continue
    d=ImageChops.difference(x,y); bb=d.getbbox()
    px=sum(1 for p in d.getdata() if p!=(0,0,0)); mx=max(max(p) for p in d.getdata())
    print(f, "px", px, "max", mx, "bbox", bb)
