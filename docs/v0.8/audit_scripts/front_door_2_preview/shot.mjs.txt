// node shot.mjs <jobs.json> <outdir> <port>: each job {name, q, w, h?, wait?, measure?: selector, click?: js}
import { chromium } from "/private/tmp/claude-501/-Users-guy--t3-worktrees-vicdata-public-t3-5896de89/d950feaa-4d45-4e5e-ba28-963a13654342/scratchpad/shots/node_modules/playwright-core/index.mjs";
import { readFileSync, writeFileSync, mkdirSync } from "node:fs";
const [jobsFile, outDir, port] = process.argv.slice(2);
mkdirSync(outDir, { recursive: true });
const jobs = JSON.parse(readFileSync(jobsFile, "utf8"));
const browser = await chromium.launch({ executablePath: "/private/tmp/claude-501/-Users-guy--t3-worktrees-vicdata-public-t3-5896de89/d950feaa-4d45-4e5e-ba28-963a13654342/scratchpad/browsers/chromium_headless_shell-1148/chrome-mac/headless_shell" });
const results = [];
for (const j of jobs) {
  const w = j.w, phone = w < 600;
  const ctx = await browser.newContext({ viewport: { width: w, height: j.h ?? (phone ? 844 : 1000) }, deviceScaleFactor: phone ? 2 : 1, isMobile: phone, hasTouch: phone });
  const page = await ctx.newPage();
  const logs = [];
  page.on("pageerror", (e) => logs.push("EXC " + String(e.message).slice(0, 300)));
  await page.goto(`http://127.0.0.1:${port}/index.html?${j.q}`);
  await page.waitForTimeout(j.wait ?? 3500);
  if (j.click) { await page.evaluate(j.click).catch((e) => logs.push("CLICK " + e.message)); await page.waitForTimeout(1200); }
  const m = await page.evaluate((sel) => {
    const doc = document.documentElement;
    const out = { scrollWidth: Math.max(doc.scrollWidth, document.body.scrollWidth), width: window.innerWidth, small: [], misses: window.__MISSES, title: document.title };
    const root = sel ? document.querySelector(sel) : document.body;
    if (root) for (const el of root.querySelectorAll("a[href], button, select, summary, input:not([type=hidden]), [role=tab], [role=switch]")) {
      const r = el.getBoundingClientRect();
      if (!r.width || !r.height) continue;
      const style = getComputedStyle(el);
      if (style.visibility === "hidden" || style.display === "none") continue;
      let box = r;
      const lab = el.closest("label");
      if ((el.tagName === "INPUT" || el.getAttribute("role") === "switch") && lab) box = lab.getBoundingClientRect();
      const host = el.closest("li > button, li > a");
      if (Math.min(box.height, Math.max(box.width, 44)) < 43.5) out.small.push(`${el.tagName.toLowerCase()} "${(el.getAttribute("aria-label") || el.textContent || "").trim().slice(0, 40)}" ${Math.round(box.width)}x${Math.round(box.height)}`);
    }
    return out;
  }, j.measure ?? null);
  await page.screenshot({ path: `${outDir}/${j.name}.png`, fullPage: true });
  results.push({ name: j.name, ...m, logs });
  console.log(j.name, "scroll", m.scrollWidth, "/", m.width, "small", m.small.length, "misses", (m.misses || []).length, m.title, logs.length ? logs[0] : "");
  await ctx.close();
}
writeFileSync(`${outDir}/measure.json`, JSON.stringify(results, null, 1));
await browser.close();
