// Teacher and Admissions homes, main vs branch: #teacher-root's DOM and a full-page PNG, byte for byte.
import { spawn } from "node:child_process";
import { writeFileSync, mkdirSync, readFileSync } from "node:fs";
const sleep = (ms) => new Promise((r) => setTimeout(r, ms));
const port = 9890;
const chrome = spawn("/Applications/Google Chrome.app/Contents/MacOS/Google Chrome", ["--headless=new", `--remote-debugging-port=${port}`, "--no-first-run", "--disable-gpu", "--hide-scrollbars", "--force-color-profile=srgb", "--font-render-hinting=none", `--user-data-dir=/tmp/homecmp-${Date.now()}`, "about:blank"], { stdio: "ignore" });
let target;
for (let i = 0; i < 60 && !target; i++) { await sleep(200); try { target = (await (await fetch(`http://127.0.0.1:${port}/json/list`)).json()).find((t) => t.type === "page"); } catch {} }
const ws = new WebSocket(target.webSocketDebuggerUrl);
await new Promise((r) => (ws.onopen = r));
let id = 0; const pend = new Map();
ws.onmessage = (m) => { const d = JSON.parse(m.data); if (d.id && pend.has(d.id)) { pend.get(d.id)(d); pend.delete(d.id); } };
const send = (method, params = {}) => new Promise((r) => { const i = ++id; pend.set(i, r); ws.send(JSON.stringify({ id: i, method, params })); });
await send("Page.enable"); await send("Runtime.enable");
mkdirSync("homecmp", { recursive: true });
const cases = [];
for (const role of ["teacher", "multi", "admin"]) for (const lens of ["teacher", "admissions"]) for (const theme of ["dark", "light"]) for (const w of [1440, 390]) if (!(role === "teacher" && lens === "admissions")) cases.push({ role, lens, theme, w });
let same = 0; const diffs = [];
for (const c of cases) {
  const out = {};
  for (const [tree, bundle, css] of [["main", "home_main", "hc_main"], ["br", "home_br", "hc_br"]]) {
    await send("Emulation.setDeviceMetricsOverride", { width: c.w, height: 900, deviceScaleFactor: 1, mobile: c.w < 500 });
    await send("Page.navigate", { url: `http://127.0.0.1:8997/index.html?case=137625&page=home&role=${c.role}&lens=${c.lens}&theme=${c.theme}&renderer=config&bundle=${bundle}&css=${css}` });
    await sleep(3500);
    const dom = (await send("Runtime.evaluate", { expression: "document.getElementById('teacher-root')?.outerHTML ?? document.body.innerHTML", returnByValue: true })).result.result.value;
    const png = (await send("Page.captureScreenshot", { format: "png", captureBeyondViewport: true })).result.data;
    out[tree] = { dom, png };
    writeFileSync(`homecmp/${c.role}_${c.lens}_${c.theme}_${c.w}_${tree}.png`, Buffer.from(png, "base64"));
  }
  const ok = out.main.dom === out.br.dom && out.main.png === out.br.png;
  if (ok) same++; else diffs.push(`${c.role}/${c.lens}/${c.theme}/${c.w}: dom ${out.main.dom === out.br.dom ? "same" : "DIFF"}, png ${out.main.png === out.br.png ? "same" : "DIFF"}`);
}
console.log(`homes: ${cases.length} pairs, identical ${same}`);
for (const d of diffs) console.log(" ", d);
chrome.kill();
process.exit(0);
