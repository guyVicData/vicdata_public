// Client timings in headless Chrome over CDP (routes stubbed with real fixtures, so this is the
// page's own draw time; the warm route times are measured separately on the server functions).
import { spawn } from "node:child_process";
const port = 9701;
const chrome = spawn("/Applications/Google Chrome.app/Contents/MacOS/Google Chrome", ["--headless=new", `--remote-debugging-port=${port}`, "--disable-gpu", "--user-data-dir=/tmp/admprev-timing", "about:blank"], { stdio: "ignore" });
const sleep = (ms) => new Promise((r) => setTimeout(r, ms));
let t; for (let i = 0; i < 50 && !t; i++) { await sleep(200); try { t = (await (await fetch(`http://127.0.0.1:${port}/json/list`)).json()).find((x) => x.type === "page"); } catch {} }
const ws = new WebSocket(t.webSocketDebuggerUrl); await new Promise((r) => (ws.onopen = r));
let id = 0; const pend = new Map(); ws.onmessage = (m) => { const d = JSON.parse(m.data); if (pend.has(d.id)) { pend.get(d.id)(d); pend.delete(d.id); } };
const send = (method, params = {}) => new Promise((r) => { const i = ++id; pend.set(i, r); ws.send(JSON.stringify({ id: i, method, params })); });
const ev = async (e) => (await send("Runtime.evaluate", { expression: e, returnByValue: true })).result.result.value;
const until = async (cond, max = 15000) => { const s = Date.now(); while (Date.now() - s < max) { if (await ev(cond)) return Date.now() - s; await sleep(10); } return -1; };
await send("Emulation.setDeviceMetricsOverride", { width: 1440, height: 1180, deviceScaleFactor: 1, mobile: false });
const url = "http://127.0.0.1:8977/index.html?case=137625&role=lead&saved=1&theme=dark";
const res = [];
for (let run = 0; run < 3; run++) {
  await send("Page.navigate", { url: "about:blank" }); await sleep(300);
  const s = Date.now();
  await send("Page.navigate", { url });
  const first = await until(`!!document.querySelector('[data-adm-panel$=".c2.now"] svg path') && !!document.querySelector('[data-adm-panel$=".c3.trends"] svg path')`);
  const firstMs = first < 0 ? -1 : Date.now() - s;
  const s2 = Date.now();
  await ev(`[...document.querySelectorAll('button')].find((b) => b.textContent === '16+').click()`);
  const sw = await until(`document.body.innerText.includes('Year 12 entry')`);
  await ev(`document.querySelector('[aria-label="Funnel"]').click()`);
  const lv = await until(`document.body.innerText.includes('From births to your')`);
  res.push({ firstDraw: firstMs, entrySwitch: sw, largeViewSwitch: lv });
  void s2;
}
console.log(JSON.stringify(res));
ws.close(); chrome.kill(); process.exit(0);
